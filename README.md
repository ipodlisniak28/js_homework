// 1. 'number' + 3 + 3
console.log('number' + 3 + 3); 
/* Результат: 'number33'
   Пояснення: Операції йдуть зліва направо. 
   1) 'number' + 3 -> 'number3' (рядок перемагає, відбувається конкатенація/злипання).
   2) 'number3' + 3 -> 'number33' (знову конкатенація рядка з числом). */

// 2. null + 3
console.log(null + 3); 
/* Результат: 3
   Пояснення: При додаванні з числом null перетворюється на 0.
   0 + 3 = 3. */

// 3. 5 && "qwerty"
console.log(5 && "qwerty"); 
/* Результат: "qwerty"
   Пояснення: Оператор && (І) шукає перше "брехливе" (false) значення. 
   Якщо все true, він повертає ОСТАННЄ значення.
   5 - це true. "qwerty" - це true. Тому повертається "qwerty". */

// 4. +'40' + +'2' + "hillel"
console.log(+'40' + +'2' + "hillel"); 
/* Результат: '42hillel'
   Пояснення: Унарний плюс (+) перед рядком перетворює його на число.
   1) +'40' стає числом 40.
   2) +'2' стає числом 2.
   3) 40 + 2 = 42 (математичне додавання).
   4) 42 + "hillel" -> '42hillel' (конкатенація числа з рядком). */

// 5. '10' - 5 === 6
console.log('10' - 5 === 6); 
/* Результат: false
   Пояснення: 
   1) '10' - 5 -> Рядок '10' перетворюється на число 10 (бо мінус - це математика). 10 - 5 = 5.
   2) 5 === 6 -> Порівняння числа 5 і 6. Це брехня (false). */

// 6. true + false
console.log(true + false); 
/* Результат: 1
   Пояснення: При математичних діях true стає 1, а false стає 0.
   1 + 0 = 1. */

// 7. '4px' - 3
console.log('4px' - 3); 
/* Результат: NaN
   Пояснення: Рядок '4px' не можна повністю перетворити на число через літери "px".
   Результат перетворення - NaN (Not a Number).
   NaN - 3 = NaN (будь-яка математика з NaN дає NaN). */

// 8. '4' - 3
console.log('4' - 3); 
/* Результат: 1
   Пояснення: Рядок '4' успішно перетворюється на число 4 (через мінус).
   4 - 3 = 1. */

// 9. '6' + 3 ** 0
console.log('6' + 3 ** 0); 
/* Результат: '61'
   Пояснення: Пріоритет піднесення до степеня (**) вищий за додавання.
   1) 3 ** 0 = 1 (будь-що в степені 0 дорівнює 1).
   2) '6' + 1 -> '61' (конкатенація). */

// 10. 12 / '6'
console.log(12 / '6'); 
/* Результат: 2
   Пояснення: Ділення (/) перетворює рядки на числа.
   '6' стає 6. 12 / 6 = 2. */

// 11. '10' + (5 === 6)
console.log('10' + (5 === 6)); 
/* Результат: '10false'
   Пояснення: Спочатку виконується дія в дужках.
   1) 5 === 6 -> false.
   2) '10' + false -> '10false' (конкатенація рядка з булевим значенням). */

// 12. null == ''
console.log(null == ''); 
/* Результат: false
   Пояснення: null при нестрогому порівнянні (==) дорівнює тільки undefined або самому собі.
   Він не дорівнює 0 або порожньому рядку. */

// 13. 3 ** (9 / 3)
console.log(3 ** (9 / 3)); 
/* Результат: 27
   Пояснення: Спочатку дужки.
   1) 9 / 3 = 3.
   2) 3 ** 3 (3 в кубі) = 3 * 3 * 3 = 27. */

// 14. !!'false' == !!'true'
console.log(!!'false' == !!'true'); 
/* Результат: true
   Пояснення: !! (подвійне заперечення) перетворює значення на булевий тип.
   1) 'false' - це непорожній рядок, отже це true.
   2) 'true' - це теж непорожній рядок, отже це true.
   3) true == true -> true. */

// 15. 0 || '0' && 1
console.log(0 || '0' && 1); 
/* Результат: 1
   Пояснення: && має вищий пріоритет за ||.
   1) '0' && 1 -> '0' це true (рядок не порожній), тому повертаємо останнє -> 1.
   2) Тепер вираз виглядає як: 0 || 1.
   3) Оператор || (АБО) шукає перше true. 0 - це false. 1 - це true. Повертає 1. */

// 16. (+null == false) < 1
console.log((+null == false) < 1); 
/* Результат: false
   Пояснення:
   1) +null стає 0.
   2) 0 == false -> true (при нестрогому порівнянні 0 дорівнює false).
   3) true < 1 -> true перетворюється на 1.
   4) 1 < 1 -> false (одиниця не менша за одиницю). */

// 17. false && true || true
console.log(false && true || true); 
/* Результат: true
   Пояснення: && виконується раніше.
   1) false && true -> false (знайшли брехню одразу).
   2) false || true -> true (шукаємо правду, знайшли її в кінці). */

// 18. false && (false || true)
console.log(false && (false || true)); 
/* Результат: false
   Пояснення: Спочатку дужки.
   1) (false || true) -> true.
   2) false && true -> false (бо перша частина false). */

// 19. (+null == false) < 1 ** 5
console.log((+null == false) < 1 ** 5); 
/* Результат: false
   Пояснення:
   1) 1 ** 5 = 1.
   2) +null = 0. 0 == false -> true.
   3) true < 1 -> (як і в прикладі 16) 1 < 1 -> false. */
